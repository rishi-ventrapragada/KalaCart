# Database & Supabase
# 2.16.0, not 2.7.1 (the old pin, which the local env had drifted from) and not
# 2.31.0 (latest). 2.17.0+ pulls realtime>=2.11.7's pydantic requirement, which
# conflicts with the pinned pydantic==2.6.3 and would cascade into fastapi and
# openai. 2.16.0 is the highest version that resolves against pydantic 2.6.3.
# It passes the API key through as an opaque header -- no JWT parsing anywhere in
# the package -- so it accepts the new sb_secret_ key format as well as legacy
# service_role JWTs.
supabase==2.16.0
psycopg2-binary==2.9.9
sqlalchemy==2.0.29
alembic==1.13.1

# Auth

# Supabase access-token verification (app/core/supabase_auth.py) uses
# jwt.PyJWKClient, which is PyJWT — a different package from python-jose above.
# The AI-Features handover shipped that module without declaring this.
# >=2.6 is required for PyJWKClient(lifespan=...).
PyJWT[crypto]==2.10.1
passlib[bcrypt]==1.7.4

# HTTP / AI

# Image processing
pillow==10.3.0
# Headless build: same cv2 API without the GUI/X11 shared libraries, which are
# useless in a container and add hundreds of MB to the image.
#
# 4.12.0.88, not 4.9.0.80. The 4.9 wheel's C extension is compiled against the
# NumPy 1.x ABI (its binary references numpy.core.multiarray) but its metadata
# only says numpy>=1.23.5 with NO upper bound -- it predates NumPy 2.0 and could
# not exclude it. So a fresh install resolved numpy 2.x and cv2 died at import
# with "numpy.core.multiarray failed to import", crashing the worker on boot.
# 4.12.0.88 is built against the NumPy 2.x ABI and declares numpy>=2,<2.3.0, so
# the pair cannot drift apart again.
opencv-python-headless==4.12.0.88

# Pinned explicitly even though opencv constrains it. numpy was previously an
# unpinned transitive dependency, which is why local and production resolved
# different versions and the mismatch only appeared on a fresh build.
numpy==2.2.6

# PDF / QR generation
# All three were imported by app/ but never declared -- requirements.txt had
# drifted from the code. reportlab is imported at MODULE level in
# app/api/business_suite.py and app/services/*_pdf.py (20 sites), so its absence
# killed the worker at boot on Render. qrcode and redis are imported inside
# functions, so they failed only when those code paths ran -- quieter, but still
# broken in production.
# All three are pure-Python wheels (py3-none-any): nothing compiles on the builder.
reportlab==4.2.5          # needs pillow>=9.0.0; satisfied by pillow 10.3.0 above
qrcode[pil]==7.4.2        # [pil] extra: storefront.py calls img.save(..., format="PNG")
redis==5.0.8              # app/core/redis.py imports redis.asyncio

# Utils
gunicorn==22.0.0
